package me.kall.narutooverlay.backplane;

import me.kall.narutooverlay.config.RenderArgs;
import me.kall.narutooverlay.theme.Palette;
import net.minecraft.client.gui.GuiGraphics;
import org.jetbrains.annotations.NotNull;

public abstract class AbstractBackplane {
    public static final int PADDING_X = 4;
    public static final int PADDING_RIGHT = 3;
    public static final int PADDING_TOP = 2;
    public static final int PADDING_BOTTOM = 2;
    public static final int ACCENT_WIDTH = 2;

    public final void render(@NotNull GuiGraphics graphics, int left, int top, int width, int height, int @NotNull [] dividers, int borderColor) {
        Frame frame = frame(left, top, width, height);

        this.draw(graphics, frame.card(), frame.palette(), dividers, borderColor);
    }

    public final void renderGroup(@NotNull GuiGraphics graphics, int left, int top, int width, int height, int @NotNull [] separators, int borderColor) {
        Frame frame = frame(left, top, width, height);

        this.draw(graphics, frame.card(), frame.palette(), new int[0], borderColor);

        int separator = frame.palette().separator() | 0xFF000000;
        int start = top + 1;
        int end = top + height - 1;
        for (int y : separators) {
            if (y <= start || y >= end) continue;
            graphics.fill(left + 1, y, left + width - 1, y + 1, separator);
        }
    }

    private static @NotNull Frame frame(int left, int top, int width, int height) {
        Palette palette = RenderArgs.palette();
        boolean accent = palette.hasAccent();
        int accentExtra = accent ? ACCENT_WIDTH + 1 : 0;

        Card card = new Card(left - PADDING_X - accentExtra, top - PADDING_TOP, left + width + PADDING_RIGHT, top + height + PADDING_BOTTOM, accent, left);
        return new Frame(palette, card);
    }

    protected abstract void draw(@NotNull GuiGraphics graphics, @NotNull Card card, @NotNull Palette palette, int @NotNull [] dividers, int borderColor);

    private record Frame(@NotNull Palette palette, @NotNull Card card) {}

    protected record Card(int left, int top, int right, int bottom, boolean accent, int origin) {
        public int accentLeft() {
            return this.left + 1;
        }

        public int accentRight() {
            return this.left + 1 + ACCENT_WIDTH;
        }

        public int dividerColumn(int offset) {
            return this.origin + offset;
        }
    }
}

package me.kall.narutooverlay.backplane;

import me.kall.narutooverlay.config.RenderArgs;
import me.kall.narutooverlay.theme.Palette;
import net.minecraft.client.gui.GuiGraphics;
import org.jetbrains.annotations.NotNull;

public class DefaultBackplane extends AbstractBackplane {
    private static final int TOP_HIGHLIGHT = 0x14FFFFFF;
    private static final float CORNER_ALPHA = 0.35F;

    public static int scaleAlpha(int argb, float factor) {
        int alpha = Math.round((argb >>> 24) * Math.max(0.0F, Math.min(1.0F, factor)));
        return alpha << 24 | argb & 0xFFFFFF;
    }

    @Override
    protected void draw(@NotNull GuiGraphics graphics, @NotNull Card card, @NotNull Palette palette, int @NotNull [] dividers, int borderColor) {
        int left = card.left();
        int top = card.top();
        int right = card.right();
        int bottom = card.bottom();

        int backplane = palette.backplane();

        graphics.fill(left + 1, top, right - 1, bottom, backplane);
        graphics.fill(left, top + 1, left + 1, bottom - 1, backplane);
        graphics.fill(right - 1, top + 1, right, bottom - 1, backplane);

        graphics.fill(left + 1, top + 1, right - 1, top + 2, TOP_HIGHLIGHT);

        if (card.accent()) {
            graphics.fillGradient(card.accentLeft(), top + 1, card.accentRight(), bottom - 1, palette.accent() | 0xFF000000, RenderArgs.borderColor() | 0xFF000000);
        }

        graphics.fill(left + 1, top, right - 1, top + 1, borderColor);
        graphics.fill(left + 1, bottom - 1, right - 1, bottom, borderColor);
        graphics.fill(left, top + 1, left + 1, bottom - 1, borderColor);
        graphics.fill(right - 1, top + 1, right, bottom - 1, borderColor);

        int corner = scaleAlpha(borderColor, CORNER_ALPHA);
        graphics.fill(left, top, left + 1, top + 1, corner);
        graphics.fill(right - 1, top, right, top + 1, corner);
        graphics.fill(left, bottom - 1, left + 1, bottom, corner);
        graphics.fill(right - 1, bottom - 1, right, bottom, corner);

        int divider = RenderArgs.borderColor();
        for (int offset : dividers) {
            int x = card.dividerColumn(offset);
            graphics.fill(x, top + 1, x + 1, bottom - 1, divider);
        }
    }
}

package me.kall.narutooverlay.backplane;

import icyllis.modernui.mc.ExtendedGuiGraphics;
import me.kall.narutooverlay.config.RenderArgs;
import me.kall.narutooverlay.theme.Palette;
import net.minecraft.client.gui.GuiGraphics;
import org.jetbrains.annotations.NotNull;

public class MuiBackplane extends AbstractBackplane {
    private static final float CORNER_RADIUS = 4.0F;
    private static final float ACCENT_RADIUS = 1.0F;
    private static final float ACCENT_INSET = 2.0F;
    private static final float MIN_DIVIDER_WIDTH = 0.5F;

    @Override
    protected void draw(@NotNull GuiGraphics graphics, @NotNull Card card, @NotNull Palette palette, int @NotNull [] dividers, int borderColor) {
        ExtendedGuiGraphics extended = new ExtendedGuiGraphics(graphics);

        float cardLeft = card.left();
        float cardTop = card.top();
        float cardRight = card.right();
        float cardBottom = card.bottom();

        extended.setColor(palette.backplane());
        extended.fillRoundRect(cardLeft, cardTop, cardRight, cardBottom, CORNER_RADIUS);

        if (card.accent()) {
            extended.setGradient(ExtendedGuiGraphics.Orientation.TOP_BOTTOM, palette.accent() | 0xFF000000, RenderArgs.borderColor() | 0xFF000000);
            extended.fillRoundRect(card.accentLeft(), cardTop + ACCENT_INSET, card.accentRight(), cardBottom - ACCENT_INSET, ACCENT_RADIUS);
        }

        float borderWidth = RenderArgs.BACKPLANE_BORDER_WIDTH.get().floatValue();
        if (borderWidth > 0) {
            float halfBorder = borderWidth / 2;
            extended.setColor(borderColor);
            extended.setStrokeWidth(borderWidth);
            extended.strokeRoundRect(cardLeft - halfBorder, cardTop - halfBorder, cardRight + halfBorder, cardBottom + halfBorder, CORNER_RADIUS + halfBorder);
        }

        if (dividers.length == 0) return;

        float dividerWidth = Math.max(borderWidth, MIN_DIVIDER_WIDTH);
        float halfDivider = dividerWidth / 2;

        extended.setColor(RenderArgs.borderColor());
        for (int offset : dividers) {
            float center = card.dividerColumn(offset) + 0.5F;
            extended.fillRoundRect(center - halfDivider, cardTop, center + halfDivider, cardBottom, halfDivider);
        }
    }
}

package me.kall.narutooverlay.card;

import it.unimi.dsi.fastutil.objects.Object2ObjectLinkedOpenHashMap;
import it.unimi.dsi.fastutil.objects.Object2ObjectOpenHashMap;
import me.kall.narutooverlay.backplane.BackplaneRenderer;
import me.kall.narutooverlay.config.ConfigSaver;
import me.kall.narutooverlay.config.ConfigVersion;
import me.kall.narutooverlay.config.RenderArgs;
import me.kall.narutooverlay.text.TextLine;
import net.minecraft.client.gui.Font;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.Minecraft;
import net.minecraftforge.common.ForgeConfigSpec;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;
import org.jetbrains.annotations.Unmodifiable;

import java.util.*;

public abstract class Card<D> {
    private final String id;
    private final int defaultLeft;
    private final int defaultTop;

    private final ForgeConfigSpec.BooleanValue enabled;
    private final ForgeConfigSpec.IntValue left;
    private final ForgeConfigSpec.IntValue top;
    private final ForgeConfigSpec.ConfigValue<String> group;
    private final ForgeConfigSpec.ConfigValue<List<? extends String>> order;

    private final Map<String, Field<D>> fields = new Object2ObjectLinkedOpenHashMap<>();
    private final Map<String, FieldConfig> configs = new Object2ObjectOpenHashMap<>();

    private @Nullable D data;
    private @Nullable TextLine line;
    private int builtVersion = -1;

    protected Card(ForgeConfigSpec.@NotNull Builder builder, @NotNull String id, int defaultLeft, int defaultTop, boolean enabledByDefault, @NotNull List<Field<D>> fields) {
        this.id = id;
        this.defaultLeft = defaultLeft;
        this.defaultTop = defaultTop;

        for (Field<D> field : fields) {
            if (this.fields.put(field.id(), field) != null) {
                throw new IllegalArgumentException("Duplicate field id '" + field.id() + "' in card " + id);
            }
        }

        builder.push(id);

        this.enabled = builder
                .comment("Whether the " + id + " overlay is shown.")
                .define("enabled", enabledByDefault);
        this.left = builder
                .comment("Horizontal position of the " + id + " overlay.")
                .defineInRange("left", defaultLeft, 0, Integer.MAX_VALUE);
        this.top = builder
                .comment("Vertical position of the " + id + " overlay.")
                .defineInRange("top", defaultTop, 0, Integer.MAX_VALUE);
        this.group = builder
                .comment(
                        "Non-empty group id shared by cards merged into one outline.",
                        "Managed by Shift-clicking cards in the chat screen; leave empty for a standalone card."
                )
                .define("group", "");
        this.order = builder
                .comment(
                        "Order of the fields, from left to right.",
                        "Available fields: " + String.join(", ", this.fields.keySet()) + ".",
                        "Unknown entries are ignored; fields that are not listed are appended in their default order."
                )
                .defineListAllowEmpty("order", new ArrayList<>(this.fields.keySet()), String.class::isInstance);

        builder.push("Fields");
        for (Field<D> field : this.fields.values()) {
            builder.push(field.id());
            this.configs.put(field.id(), new FieldConfig(builder, field.hasLabel(), field.hasUnit(), field.enabledByDefault()));
            builder.pop();
        }
        builder.pop();

        builder.pop();

        CardRegistry.register(this);
    }

    public final @NotNull String id() {
        return this.id;
    }

    public final int left() {
        return this.left.get();
    }

    public final int top() {
        return this.top.get();
    }

    public final boolean active() {
        if (!this.enabled.get()) return false;

        for (FieldConfig config : this.configs.values()) {
            if (config.enabled()) return true;
        }

        return false;
    }

    protected final void accept(@NotNull D data) {
        this.data = data;
        this.builtVersion = -1;
    }

    public final void moveTo(double left, double top) {
        this.moveTo(left, top, null);
    }

    public final void moveTo(double left, double top, @Nullable Runnable extraSync) {
        this.left.set(Math.max(0, (int) left));
        this.top.set(Math.max(0, (int) top));
        if (extraSync != null) extraSync.run();
        ConfigSaver.changed();
    }

    public final void resetPosition() {
        this.resetPosition(null);
    }

    public final void resetPosition(@Nullable Runnable extraSync) {
        this.left.set(this.defaultLeft);
        this.top.set(this.defaultTop);
        if (extraSync != null) extraSync.run();
    }

    public final void resetAll() {
        this.resetPosition();
        this.enabled.set(this.enabled.getDefault());
        this.group.set("");
        this.order.set(new ArrayList<>(this.fields.keySet()));
        for (FieldConfig config : this.configs.values()) config.reset();
        ConfigVersion.bump();
        ConfigSaver.changed();
    }

    public final @NotNull String group() {
        return this.group.get();
    }

    public final void setGroup(@NotNull String group) {
        this.group.set(group);
    }

    public final boolean contains(double mouseX, double mouseY) {
        TextLine own = this.line();
        if (own == null) return false;

        double left = this.left.get();
        double top = this.top.get();
        double width = this.width(own);
        double height = Minecraft.getInstance().font.lineHeight;

        return mouseX >= left && mouseX <= left + width && mouseY >= top && mouseY <= top + height;
    }

    final void render(@NotNull GuiGraphics graphics) {
        TextLine line = this.line();
        if (line == null) return;

        Font font = Minecraft.getInstance().font;

        int left = this.left.get();
        int top = this.top.get();
        int width = this.width(line);
        int height = font.lineHeight;

        int[] dividers = line.dividers();

        if (RenderArgs.DRAW_BACKPLANE.get()) {
            int border = CardRegistry.isSelected(this) ? 0xFFFFC42E : RenderArgs.borderColor();
            BackplaneRenderer.render(graphics, left, top, width, height, dividers, border);
        } else {
            this.renderDividers(graphics, font);
        }

        line.draw(graphics, font, left, top, RenderArgs.DROP_SHADOW.get());
    }

    final void renderLine(@NotNull GuiGraphics graphics, @NotNull Font font) {
        TextLine line = this.line();
        if (line == null) return;

        line.draw(graphics, font, this.left.get(), this.top.get(), RenderArgs.DROP_SHADOW.get());
    }

    final void renderDividers(@NotNull GuiGraphics graphics, @NotNull Font font) {
        TextLine line = this.line();
        if (line == null) return;

        int left = this.left.get();
        int top = this.top.get();
        int color = RenderArgs.DRAW_BACKPLANE.get()
                ? RenderArgs.borderColor()
                : RenderArgs.palette().separator() | 0xFF000000;

        for (int offset : line.dividers()) {
            int x = left + offset;
            graphics.fill(x, top + 1, x + 1, top + font.lineHeight - 1, color);
        }
    }

    private int width(@NotNull TextLine own) {
        return RenderArgs.EQUAL_WIDTH.get() ? Math.max(own.width(), CardRegistry.sharedWidth()) : own.width();
    }

    final @Nullable TextLine line() {
        if (!this.enabled.get()) return null;

        D data = this.data;
        if (data == null) return null;

        int version = ConfigVersion.get();
        if (this.builtVersion != version) {
            this.line = this.build(data);
            this.builtVersion = version;
        }

        return this.line;
    }

    private @Nullable TextLine build(@NotNull D data) {
        TextLine.Builder builder = TextLine.builder(Minecraft.getInstance().font);
        boolean first = true;

        for (Field<D> field : this.orderedFields()) {
            FieldConfig config = this.configs.get(field.id());
            if (!config.enabled()) continue;

            if (!first) builder.divider();
            first = false;

            field.append(builder, data, config);
        }

        return first ? null : builder.build();
    }

    private @NotNull List<Field<D>> orderedFields() {
        Set<String> ids = new LinkedHashSet<>();
        for (String id : this.order.get()) {
            if (this.fields.containsKey(id)) ids.add(id);
        }
        ids.addAll(this.fields.keySet());

        List<Field<D>> ordered = new ArrayList<>(ids.size());
        for (String id : ids) ordered.add(this.fields.get(id));
        return ordered;
    }

    public final ForgeConfigSpec.@NotNull BooleanValue enabledValue() {
        return this.enabled;
    }

    public final void setEnabled(boolean enabled) {
        this.enabledValue().set(enabled);
        ConfigSaver.changed();
    }

    public final ForgeConfigSpec.@NotNull IntValue leftValue() {
        return this.left;
    }

    public final void setLeft(int left) {
        this.leftValue().set(left);
        ConfigSaver.changed();
    }

    public final ForgeConfigSpec.@NotNull IntValue topValue() {
        return this.top;
    }

    public final void setTop(int top) {
        this.topValue().set(top);
        ConfigSaver.changed();
    }

    public final @NotNull @Unmodifiable List<Field<D>> allFields() {
        return List.copyOf(this.fields.values());
    }

    public final @NotNull FieldConfig fieldConfig(@NotNull String fieldId) {
        FieldConfig config = this.configs.get(fieldId);
        if (config == null) throw new IllegalArgumentException("Unknown field '" + fieldId + "' in card " + this.id);
        return config;
    }

    public final int lineWidth() {
        TextLine own = this.line();
        return own == null ? 0 : this.width(own);
    }
}

package me.kall.narutooverlay.card;

import it.unimi.dsi.fastutil.objects.ObjectArrayList;
import it.unimi.dsi.fastutil.objects.ObjectOpenHashSet;
import me.kall.narutooverlay.backplane.AbstractBackplane;
import me.kall.narutooverlay.backplane.BackplaneRenderer;
import me.kall.narutooverlay.config.RenderArgs;
import me.kall.narutooverlay.text.TextLine;
import net.minecraft.client.gui.Font;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.Minecraft;
import net.minecraftforge.client.event.RenderGuiOverlayEvent;
import net.minecraftforge.client.gui.overlay.VanillaGuiOverlay;
import org.jetbrains.annotations.Contract;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;
import org.jetbrains.annotations.UnmodifiableView;

import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public final class CardRegistry {
    private static final List<Card<?>> CARDS = new ObjectArrayList<>(3);
    private static final int GROUP_ROW_GAP = 4;

    private static @Nullable Card<?> selected;

    private CardRegistry() {}

    static void register(@NotNull Card<?> card) {
        CARDS.add(card);
    }

    @Contract(pure = true)
    public static @NotNull @UnmodifiableView List<Card<?>> all() {
        return Collections.unmodifiableList(CARDS);
    }

    public static void render(@NotNull RenderGuiOverlayEvent.Post event) {
        if (event.getOverlay() != VanillaGuiOverlay.HOTBAR.type()) return;

        GuiGraphics graphics = event.getGuiGraphics();
        Set<String> drawnGroups = new HashSet<>();

        for (Card<?> card : CARDS) {
            if (!card.active()) continue;

            String group = card.group();
            if (!group.isEmpty()) {
                if (!drawnGroups.add(group)) continue;

                List<Card<?>> members = activeMembers(group);
                if (members.size() > 1) {
                    renderGroup(graphics, members);
                    continue;
                }
            }

            card.render(graphics);
        }
    }

    private static @NotNull List<Card<?>> activeMembers(@NotNull String group) {
        List<Card<?>> members = new ArrayList<>(2);
        for (Card<?> card : CARDS) {
            if (card.active() && group.equals(card.group())) members.add(card);
        }
        return members;
    }

    public static @NotNull List<Card<?>> groupMembers(@NotNull Card<?> card) {
        String group = card.group();
        if (group.isEmpty()) return List.of(card);

        List<Card<?>> members = new ArrayList<>(2);
        for (Card<?> candidate : CARDS) {
            if (group.equals(candidate.group())) members.add(candidate);
        }
        return members;
    }

    private static void renderGroup(@NotNull GuiGraphics graphics, @NotNull List<Card<?>> members) {
        members.sort(Comparator.comparingInt((Card<?> card) -> card.top()).thenComparingInt(Card::left));

        Font font = Minecraft.getInstance().font;
        Bounds bounds = bounds(members);

        int[] separators = new int[Math.max(0, members.size() - 1)];
        for (int i = 1; i < members.size(); i++) {
            separators[i - 1] = members.get(i).top() - 2;
        }

        if (RenderArgs.DRAW_BACKPLANE.get()) {
            int border = members.stream().anyMatch(CardRegistry::isSelected) ? 0xFFFFC42E : RenderArgs.borderColor();
            BackplaneRenderer.renderGroup(graphics, bounds.left(), bounds.top(),
                    bounds.right() - bounds.left(), bounds.bottom() - bounds.top(), separators, border);
        } else {
            int color = RenderArgs.palette().separator() | 0xFF000000;
            for (int y : separators) {
                if (y <= bounds.top() || y >= bounds.bottom()) continue;
                graphics.fill(bounds.left(), y, bounds.right(), y + 1, color);
            }
        }

        for (Card<?> card : members) card.renderDividers(graphics, font);
        for (Card<?> card : members) card.renderLine(graphics, font);
    }

    public static @Nullable Card<?> pick(double mouseX, double mouseY) {
        Set<String> checkedGroups = new ObjectOpenHashSet<>();

        for (int i = CARDS.size() - 1; i >= 0; i--) {
            Card<?> card = CARDS.get(i);
            if (!card.active()) continue;

            String group = card.group();
            if (!group.isEmpty()) {
                if (!checkedGroups.add(group)) continue;

                List<Card<?>> members = activeMembers(group);
                if (members.size() > 1 && groupContains(members, mouseX, mouseY)) {
                    for (Card<?> member : members) {
                        if (member.contains(mouseX, mouseY)) return member;
                    }
                    return members.get(0);
                }

                continue;
            }

            if (card.contains(mouseX, mouseY)) return card;
        }

        return null;
    }

    private static boolean groupContains(@NotNull List<Card<?>> members, double mouseX, double mouseY) {
        Bounds bounds = bounds(members);

        int accentExtra = RenderArgs.palette().hasAccent() ? AbstractBackplane.ACCENT_WIDTH + 1 : 0;
        int x0 = bounds.left() - AbstractBackplane.PADDING_X - accentExtra;
        int x1 = bounds.right() + AbstractBackplane.PADDING_RIGHT;
        int y0 = bounds.top() - AbstractBackplane.PADDING_TOP;
        int y1 = bounds.bottom() + AbstractBackplane.PADDING_BOTTOM;

        return mouseX >= x0 && mouseX <= x1 && mouseY >= y0 && mouseY <= y1;
    }

    private static @NotNull Bounds bounds(@NotNull List<Card<?>> members) {
        Font font = Minecraft.getInstance().font;
        int left = Integer.MAX_VALUE;
        int top = Integer.MAX_VALUE;
        int right = Integer.MIN_VALUE;
        int bottom = Integer.MIN_VALUE;

        for (Card<?> card : members) {
            left = Math.min(left, card.left());
            top = Math.min(top, card.top());
            right = Math.max(right, card.left() + card.lineWidth());
            bottom = Math.max(bottom, card.top() + font.lineHeight);
        }

        return new Bounds(left, top, right, bottom);
    }

    private record Bounds(int left, int top, int right, int bottom) {}

    private static int leftEdge(@NotNull List<Card<?>> members) {
        int left = Integer.MAX_VALUE;
        for (Card<?> member : members) left = Math.min(left, member.left());
        return left;
    }

    private static int bottomEdge(@NotNull List<Card<?>> members, int lineHeight) {
        int bottom = Integer.MIN_VALUE;
        for (Card<?> member : members) bottom = Math.max(bottom, member.top() + lineHeight);
        return bottom;
    }

    public static boolean merge(@NotNull Card<?> first, @NotNull Card<?> second) {
        if (first == second || !first.active() || !second.active()) return false;

        String firstGroup = first.group();
        String secondGroup = second.group();
        if (!firstGroup.isEmpty() && firstGroup.equals(secondGroup)) return false;

        Font font = Minecraft.getInstance().font;

        if (!firstGroup.isEmpty() && !secondGroup.isEmpty()) {
            List<Card<?>> source = activeMembers(firstGroup);
            List<Card<?>> target = activeMembers(secondGroup);
            if (source.isEmpty() || target.isEmpty()) return false;

            int left = leftEdge(source);
            int sourceBottom = bottomEdge(source, font.lineHeight);

            int targetTop = Integer.MAX_VALUE;
            for (Card<?> member : target) targetTop = Math.min(targetTop, member.top());

            for (Card<?> member : target) {
                member.moveTo(left, sourceBottom + GROUP_ROW_GAP + (member.top() - targetTop));
                member.setGroup(firstGroup);
            }

            return true;
        }

        if (firstGroup.isEmpty() && secondGroup.isEmpty()) {
            int left = Math.min(first.left(), second.left());
            int top = Math.min(first.top(), second.top());

            first.moveTo(left, top);
            second.moveTo(left, top + font.lineHeight + GROUP_ROW_GAP);

            String group = "g:" + first.id() + "+" + second.id();
            first.setGroup(group);
            second.setGroup(group);
            return true;
        }

        if (!firstGroup.isEmpty()) {
            List<Card<?>> source = activeMembers(firstGroup);
            if (source.isEmpty()) return false;

            int sourceTop = Integer.MAX_VALUE;
            for (Card<?> member : source) sourceTop = Math.min(sourceTop, member.top());

            int left = second.left();
            int top = second.top();
            int offset = top + font.lineHeight + GROUP_ROW_GAP - sourceTop;

            for (Card<?> member : source) member.moveTo(left, member.top() + offset);

            second.setGroup(firstGroup);
            return true;
        }

        List<Card<?>> existing = activeMembers(secondGroup);
        if (existing.isEmpty()) return false;

        int left = leftEdge(existing);
        int bottom = bottomEdge(existing, font.lineHeight);

        first.moveTo(left, bottom + GROUP_ROW_GAP);
        first.setGroup(secondGroup);
        return true;
    }

    public static boolean split(@NotNull Card<?> card) {
        String group = card.group();
        if (group.isEmpty()) return false;

        List<Card<?>> members = groupMembers(card);
        members.sort(Comparator.comparingInt((Card<?> member) -> member.top()).thenComparingInt(Card::left));

        if (members.isEmpty()) return false;

        Font font = Minecraft.getInstance().font;
        int top = members.get(0).top();

        for (Card<?> member : members) {
            member.setGroup("");
            member.moveTo(member.left(), top);
            top += font.lineHeight + GROUP_ROW_GAP + 2;
        }

        return members.size() > 1;
    }

    public static void select(@Nullable Card<?> card) {
        selected = card;
    }

    public static @Nullable Card<?> selected() {
        return selected;
    }

    public static boolean isSelected(@Nullable Card<?> card) {
        return card != null && card == selected;
    }

    static int sharedWidth() {
        int width = 0;
        for (Card<?> card : CARDS) {
            TextLine line = card.line();
            if (line != null) width = Math.max(width, line.width());
        }
        return width;
    }

    public static void resetPositions() {
        selected = null;
        for (Card<?> card : CARDS) {
            card.setGroup("");
            card.resetPosition();
        }
    }

    public static void resetAll() {
        selected = null;
        for (Card<?> card : CARDS) card.resetAll();
    }
}

import net.minecraft.Util;

public final class ConfigSaver {
    private static final long SAVE_DELAY_MS = 400L;

    private static boolean dirty;
    private static long lastChange;

    private ConfigSaver() {}

    public static void changed() {
        ConfigVersion.bump();
        dirty = true;
        lastChange = Util.getMillis();
    }

    public static void tick() {
        if (dirty && Util.getMillis() - lastChange >= SAVE_DELAY_MS) {
            flush();
        }
    }

    public static void flush() {
        if (!dirty) return;
        dirty = false;
        RenderArgs.CONFIG.save();
    }
}

    public static <T> void set(ForgeConfigSpec.@NotNull ConfigValue<T> config, T value) {
        config.set(value);
        ConfigSaver.changed();
    }

    public static @NotNull Palette palette() {
        return THEME.get().palette();
    }

    public static int borderColor() {
        if (BACKPLANE_BORDER_OVERRIDE.get()) {
            return BACKPLANE_BORDER_ALPHA.get() << 24 | BACKPLANE_BORDER_RED.get() << 16 | BACKPLANE_BORDER_GREEN.get() << 8 | BACKPLANE_BORDER_BLUE.get();
        }

        return palette().border();
    }

package me.kall.narutooverlay.config.ui.background;

import me.kall.narutooverlay.config.RenderArgs;
import net.minecraftforge.fml.loading.FMLPaths;
import org.jetbrains.annotations.NotNull;

import java.io.IOException;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class NarutoJarResources {
    public static final String FOLDER = RenderArgs.DEFAULT_BACKGROUND_FOLDER;
    public static final List<String> NAMES = RenderArgs.DEFAULT_BACKGROUND_ORDER;

    private static final List<Asset> ASSETS = List.of(
            new Asset("bg1.jpg", "/assets/narutooverlay/textures/bg1.jpg"),
            new Asset("bg2.jpg", "/assets/narutooverlay/textures/bg2.jpg"),
            new Asset("logo.jpg", "/logo.jpg")
    );

    private record Asset(@NotNull String name, @NotNull String resource) {}

    public static void install() {
        Path folder = folder();
        try {
            Files.createDirectories(folder);
        } catch (IOException ignored) {
            return;
        }

        for (Asset asset : ASSETS) copy(asset, folder.resolve(asset.name()));
    }

    public static @NotNull Path folder() {
        return FMLPaths.GAMEDIR.get().resolve(FOLDER);
    }

    public static @NotNull List<Path> files() {
        List<Path> paths = new ArrayList<>(NAMES.size());
        for (String name : NAMES) {
            Path path = folder().resolve(name);
            if (Files.isRegularFile(path)) paths.add(path);
        }
        return paths;
    }

    public static boolean available() {
        return !files().isEmpty();
    }

    private static void copy(Asset asset, Path target) {
        if (Files.exists(target)) return;

        try (InputStream input = NarutoJarResources.class.getResourceAsStream(asset.resource())) {
            if (input == null) return;
            Files.copy(input, target);
        } catch (IOException ignored) {}
    }
}

            }
            if (absolute.toFile().isFile() && os.contains("mac")) {
                new ProcessBuilder("open", "-R", absolute.toString()).start();
                return;
            }
        } catch (Throwable ignored) {}

        try {
            Util.getPlatform().openFile(absolute.getParent().toFile());
        } catch (Throwable ignored) {}
    }

package me.kall.narutooverlay.event.api;

import it.unimi.dsi.fastutil.objects.ObjectArrayList;
import net.minecraftforge.event.TickEvent;
import net.minecraftforge.eventbus.api.Event;
import net.minecraftforge.fml.event.IModBusEvent;
import org.jetbrains.annotations.NotNull;

import java.util.function.BooleanSupplier;
import java.util.function.IntSupplier;
import java.util.List;

public final class TaskRegistryEvent extends Event implements IModBusEvent {
    public static final TaskRegistryEvent INSTANCE = new TaskRegistryEvent();

    private static final List<Task> TASKS = new ObjectArrayList<>();

    public void register(IntSupplier interval, BooleanSupplier active, Runnable task) {
        TASKS.add(new Task(task, interval, active));
    }

    public static void update(TickEvent.@NotNull ClientTickEvent event) {
        if (event.phase != TickEvent.Phase.END) return;
        for (Task task : TASKS) {
            if (task.check()) task.run();
        }
    }

    static final class Task implements Runnable {
        final Runnable task;
        final IntSupplier interval;
        final BooleanSupplier active;
        int cooldown;

        Task(Runnable task, IntSupplier interval, BooleanSupplier active) {
            this.task = task;
            this.interval = interval;
            this.active = active;
        }

        boolean check() {
            if (!this.active.getAsBoolean()) {
                this.cooldown = Integer.MAX_VALUE - 1;
                return false;
            }

            return ++this.cooldown >= this.interval.getAsInt();
        }

        @Override
        public void run() {
            this.cooldown = 0;
            this.task.run();
        }
    }
}

package me.kall.narutooverlay.event;

import me.kall.narutooverlay.config.RenderArgs;
import me.kall.narutooverlay.event.api.*;
import me.kall.narutooverlay.event.api.FpsUpdateEvent;
import me.kall.narutooverlay.event.api.TaskRegistryEvent;
import net.minecraftforge.common.MinecraftForge;
import net.minecraftforge.event.TickEvent;
import org.jetbrains.annotations.NotNull;

import java.util.ArrayDeque;
import java.util.Arrays;

public final class FpsTracker {
    private static final double NANOS = 1_000_000_000.0D;
    private static final long NANOS_PER_MILLISECOND = 1_000_000L;

    private static final ArrayDeque<FrameInterval> intervals = new ArrayDeque<>();

    private static long total = 0L;

    private static long last = 0L;

    public static void listen(@NotNull TaskRegistryEvent event) {
        event.register(RenderArgs.FPS_COOLDOWN::get, RenderArgs.FPS::active, () -> {
            pruneWindow(System.nanoTime(), RenderArgs.FPS_WINDOW.get());
            if (intervals.isEmpty()) return;

            int size = intervals.size();
            double inst = NANOS / (double) intervals.peekLast().interval;
            double avg = NANOS * (double) size / (double) total;

            if (size < 10) return;

            long[] sortedIntervals = new long[size];
            int index = 0;
            for (FrameInterval interval : intervals) sortedIntervals[index++] = interval.interval;
            Arrays.sort(sortedIntervals);
            double min = NANOS / (double) sortedIntervals[size - 1];

            int lowCount = Math.max(1, (int) Math.ceil(size * 0.01D));

            double low = 0.0D;
            for (int i = size - lowCount; i < size; i++) low += NANOS / (double) sortedIntervals[i];
            low /= lowCount;

            MinecraftForge.EVENT_BUS.post(new FpsUpdateEvent(inst, min, avg, low));
        });
    }

    public static void tick(TickEvent.@NotNull RenderTickEvent event) {
        if (event.phase != TickEvent.Phase.END) return;

        long now = System.nanoTime();

        if (last != 0L) {
            long delta = now - last;
            if (delta > 0L) {
                intervals.addLast(new FrameInterval(now, delta));
                total += delta;
            }
        }

        last = now;
        pruneWindow(now, RenderArgs.FPS_WINDOW.get());
    }

    private static void pruneWindow(long now, int windowMillis) {
        long cutoff = now - windowMillis * NANOS_PER_MILLISECOND;
        while (!intervals.isEmpty() && intervals.peekFirst().timestamp < cutoff) {
            total -= intervals.removeFirst().interval;
        }
    }

    private record FrameInterval(long timestamp, long interval) {}
}

package me.kall.narutooverlay.event;

import com.sun.management.GarbageCollectionNotificationInfo;
import com.sun.management.GcInfo;
import javax.management.NotificationEmitter;
import javax.management.openmbean.CompositeData;

import java.lang.management.GarbageCollectorMXBean;
import java.lang.management.ManagementFactory;
import java.time.Instant;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.AtomicReference;

public class GCTracker {
    private static final AtomicReference<Instant> TIME = new AtomicReference<>();
    private static final AtomicLong DURATION = new AtomicLong();
    private static final AtomicReference<String> NAME = new AtomicReference<>();

    public static void setup() {
        for (GarbageCollectorMXBean bean : ManagementFactory.getGarbageCollectorMXBeans()) {
            if (bean instanceof NotificationEmitter emitter) {
                emitter.addNotificationListener((notification, handback) -> {
                    if (notification.getType().equals(GarbageCollectionNotificationInfo.GARBAGE_COLLECTION_NOTIFICATION)) {
                        GarbageCollectionNotificationInfo info = GarbageCollectionNotificationInfo.from((CompositeData) notification.getUserData());
                        GcInfo gcInfo = info.getGcInfo();

                        TIME.set(Instant.ofEpochMilli(ManagementFactory.getRuntimeMXBean().getStartTime() + gcInfo.getStartTime()));
                        DURATION.set(gcInfo.getDuration());
                        NAME.set(info.getGcName());
                    }
                }, null, null);
            }
        }
    }

    public static Instant time() {
        return TIME.get();
    }

    public static long duration() {
        return DURATION.get();
    }

    public static String name() {
        return NAME.get();
    }

package me.kall.narutooverlay.event;

import com.mojang.blaze3d.platform.Window;
import me.kall.narutooverlay.card.Card;
import me.kall.narutooverlay.card.CardRegistry;
import me.kall.narutooverlay.config.ConfigSaver;
import net.minecraft.client.gui.screens.ChatScreen;
import net.minecraft.client.gui.screens.Screen;
import net.minecraft.client.Minecraft;
import net.minecraftforge.client.event.InputEvent;
import net.minecraftforge.event.TickEvent;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;
import org.lwjgl.glfw.GLFW;

public class TextRelocation {
    private static @Nullable Card<?> dragging;

    private static double deltaX;
    private static double deltaY;

    public static void click(InputEvent.MouseButton.@NotNull Pre event) {
        if (!isLeftMouse(event, GLFW.GLFW_PRESS)) return;
        if (!(Minecraft.getInstance().screen instanceof ChatScreen)) return;

        double mouseX = mouseX();
        double mouseY = mouseY();

        Card<?> card = pick(mouseX, mouseY);

        if (card == null) {
            if (!Screen.hasShiftDown() && !Screen.hasControlDown()) CardRegistry.select(null);
            return;
        }

        if (Screen.hasControlDown()) {
            if (CardRegistry.split(card)) ConfigSaver.changed();
            CardRegistry.select(null);
            dragging = null;
            event.setCanceled(true);
            return;
        }

        if (Screen.hasShiftDown()) {
            CardRegistry.select(CardRegistry.isSelected(card) ? null : card);
            dragging = null;
            event.setCanceled(true);
            return;
        }

        Card<?> mergeSource = CardRegistry.selected();
        if (mergeSource != null) {
            CardRegistry.select(null);
            dragging = null;

            if (mergeSource != card && CardRegistry.merge(mergeSource, card)) {
                ConfigSaver.changed();
            }

            event.setCanceled(true);
            return;
        }

        dragging = card;
        deltaX = mouseX - card.left();
        deltaY = mouseY - card.top();
    }

    public static @Nullable Card<?> pick(double mouseX, double mouseY) {
        return CardRegistry.pick(mouseX, mouseY);
    }

    public static double mouseX() {
        Minecraft minecraft = Minecraft.getInstance();
        Window window = minecraft.getWindow();
        return minecraft.mouseHandler.xpos() * window.getGuiScaledWidth() / window.getScreenWidth();
    }

    public static double mouseY() {
        Minecraft minecraft = Minecraft.getInstance();
        Window window = minecraft.getWindow();
        return minecraft.mouseHandler.ypos() * window.getGuiScaledHeight() / window.getScreenHeight();
    }

    public static void drag(TickEvent.@NotNull ClientTickEvent event) {
        if (!event.phase.equals(TickEvent.Phase.START) || dragging == null) return;

        Card<?> moving = dragging;
        int oldLeft = moving.left();
        int oldTop = moving.top();

        moving.moveTo(mouseX() - deltaX, mouseY() - deltaY);

        int dx = moving.left() - oldLeft;
        int dy = moving.top() - oldTop;
        if (dx == 0 && dy == 0) return;

        for (Card<?> member : CardRegistry.groupMembers(moving)) {
            if (member != moving) member.moveTo(member.left() + dx, member.top() + dy);
        }
    }

    public static void release(InputEvent.MouseButton.@NotNull Pre event) {
        if (isLeftMouse(event, GLFW.GLFW_RELEASE)) {
            if (dragging != null) {
                ConfigSaver.changed();
                dragging = null;
            }
        }
    }

    private static boolean isLeftMouse(InputEvent.MouseButton.@NotNull Pre event, int action) {
        return event.getButton() == GLFW.GLFW_MOUSE_BUTTON_LEFT && event.getAction() == action;
    }
}

package me.kall.narutooverlay.text;

import me.kall.narutooverlay.config.RenderArgs;

/**
 * HUD-local shimmer clock. It deliberately does not depend on the NarutoUI API, so the HUD keeps
 * working when the optional configuration-screen library is absent.
 */
final class HudShimmerClock {
    private static final double MAX_STEP_SECONDS = 0.1D;
    private static final double SMOOTHING = 10.0D;
    private static final long SCALE_DEBOUNCE_MS = 140L;
    private static final long SCALE_TRANSITION_MS = 900L;
    private static final double SCALE_EPSILON = 0.001D;

    private boolean initialized;
    private long lastMillis;
    private double time;
    private double speed = 1.0D;
    private double strength = 1.0D;
    private double scaleFrom = 1.0D;
    private double scaleTo = 1.0D;
    private double scaleMix = 1.0D;
    private double pendingScale = 1.0D;
    private long pendingSince;
    private long scaleTransitionStart;
    private boolean scaleTransition;

    synchronized void update() {
        long now = System.currentTimeMillis();

        double targetSpeed = RenderArgs.SHIMMER_SPEED.get();
        double targetScale = RenderArgs.SHIMMER_SCALE.get();
        double targetStrength = RenderArgs.SHIMMER_STRENGTH.get();

        if (!this.initialized) {
            this.initialized = true;
            this.lastMillis = now;
            this.pendingSince = now;
            this.speed = targetSpeed;
            this.strength = targetStrength;
            this.scaleFrom = targetScale;
            this.scaleTo = targetScale;
            this.pendingScale = targetScale;
            this.scaleMix = 1.0D;
            return;
        }

        double dt = Math.max(0.0D, Math.min(MAX_STEP_SECONDS, (now - this.lastMillis) / 1000.0D));
        this.lastMillis = now;

        if (dt > 0.0D) {
            double blend = 1.0D - Math.exp(-dt * SMOOTHING);
            this.speed += (targetSpeed - this.speed) * blend;
            this.strength += (targetStrength - this.strength) * blend;
            this.time += dt * this.speed;
        }

        if (Math.abs(targetScale - this.pendingScale) > SCALE_EPSILON) {
            this.pendingScale = targetScale;
            this.pendingSince = now;
        }

        if (this.scaleTransition) {
            double progress = (now - this.scaleTransitionStart) / (double) SCALE_TRANSITION_MS;
            if (progress >= 1.0D) {
                this.scaleFrom = this.scaleTo;
                this.scaleMix = 1.0D;
                this.scaleTransition = false;
            } else {
                double p = Math.max(0.0D, progress);
                this.scaleMix = p * p * (3.0D - 2.0D * p);
            }
        }

        if (!this.scaleTransition && now - this.pendingSince >= SCALE_DEBOUNCE_MS
                && Math.abs(this.pendingScale - this.scaleTo) > SCALE_EPSILON) {
            this.scaleFrom = this.scaleTo;
            this.scaleTo = this.pendingScale;
            this.scaleMix = 0.0D;
            this.scaleTransitionStart = now;
            this.scaleTransition = true;
        }
    }

    synchronized double time() {
        return this.time;
    }

    synchronized double scaleFrom() {
        return this.scaleFrom;
    }

    synchronized double scaleTo() {
        return this.scaleTo;
    }

    synchronized double scaleMix() {
        return this.scaleMix;
    }

    synchronized double strength() {
        return this.strength;
    }
}

package me.kall.narutooverlay.text;

import it.unimi.dsi.fastutil.ints.IntArrayList;
import it.unimi.dsi.fastutil.objects.ObjectArrayList;
import me.kall.narutooverlay.config.RenderArgs;
import net.minecraft.client.gui.Font;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.network.chat.Component;
import net.minecraft.network.chat.MutableComponent;
import net.minecraft.network.chat.Style;
import net.minecraft.network.chat.TextColor;
import net.minecraft.util.FormattedCharSequence;
import net.minecraft.util.FormattedCharSink;
import net.minecraft.Util;
import org.jetbrains.annotations.Contract;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.util.List;

public final class TextLine {
    public static final int DIVIDER_GAP = 7;
    private static final HudShimmerClock SHIMMER_CLOCK = new HudShimmerClock();

    public enum Align {
        LEFT,
        RIGHT
    }

    private record Run(Component text, FormattedCharSequence sequence, int x) {}

    private final Run[] runs;
    private final int[] dividers;
    private final int width;

    private final CachedSequence[] shimmerCache;
    private long shimmerCacheAt;
    private int shimmerCacheLeft = Integer.MIN_VALUE;
    private boolean shimmerCacheValid;

    @Contract(pure = true)
    private TextLine(Run @NotNull [] runs, int[] dividers, int width) {
        this.runs = runs;
        this.dividers = dividers;
        this.width = width;
        this.shimmerCache = new CachedSequence[runs.length];
    }

    public int width() {
        return this.width;
    }

    public int[] dividers() {
        return this.dividers;
    }

    public void draw(@NotNull GuiGraphics graphics, @NotNull Font font, int left, int top, boolean shadow) {
        if (RenderArgs.SHIMMER.get() && RenderArgs.palette().hasAccent()) {
            if (RenderArgs.CACHE_SHIMMER.get()) {
                this.drawShimmerCached(graphics, font, left, top, shadow);
            } else {
                this.drawShimmerLegacy(graphics, font, left, top, shadow);
            }
            return;
        }

        float y = top + textOffset();
        for (Run run : this.runs) {
            drawRun(graphics, font, run.sequence(), left + run.x(), y, shadow);
        }
    }

    private void drawShimmerCached(@NotNull GuiGraphics graphics, @NotNull Font font, int left, int top, boolean shadow) {
        long now = Util.getMillis();
        int fps = Math.max(1, RenderArgs.SHIMMER_FPS.get());
        long interval = Math.max(1L, 1000L / fps);

        if (!this.shimmerCacheValid || left != this.shimmerCacheLeft || now - this.shimmerCacheAt >= interval) {
            this.rebuildShimmerCache(left, now);
        }

        float y = top + textOffset();
        for (int i = 0; i < this.runs.length; i++) {
            CachedSequence sequence = this.shimmerCache[i];
            if (sequence == null) continue;

            drawRun(graphics, font, sequence, left + this.runs[i].x(), y, shadow);
        }
    }

    private void rebuildShimmerCache(int left, long now) {
        ShimmerState state = shimmerState();

        for (int i = 0; i < this.runs.length; i++) {
            this.shimmerCache[i] = buildShimmerCache(this.runs[i], left, state);
        }

        this.shimmerCacheAt = now;
        this.shimmerCacheLeft = left;
        this.shimmerCacheValid = true;
    }

    @Contract("_, _, _ -> new")
    private static @NotNull CachedSequence buildShimmerCache(@NotNull Run run, int left, @NotNull ShimmerState state) {
        float baseX = left + run.x();
        IntArrayList indices = new IntArrayList();
        IntArrayList codepoints = new IntArrayList();
        ObjectArrayList<Style> styles = new ObjectArrayList<>();

        run.sequence().accept((index, style, codePoint) -> {
            indices.add(index);
            codepoints.add(codePoint);
            styles.add(style.withColor(TextColor.fromRgb(shimmerColor(style, baseX, index, state))));
            return true;
        });

        return new CachedSequence(indices.toIntArray(), codepoints.toIntArray(), styles.toArray(new Style[0]));
    }

    private void drawShimmerLegacy(@NotNull GuiGraphics graphics, @NotNull Font font, int left, int top, boolean shadow) {
        ShimmerState state = shimmerState();
        float y = top + textOffset();

        for (Run run : this.runs) {
            float baseX = left + run.x();

            FormattedCharSequence shimmered = sink -> run.sequence().accept((index, style, codePoint) -> {
                int color = shimmerColor(style, baseX, index, state);
                return sink.accept(index, style.withColor(TextColor.fromRgb(color)), codePoint);
            });

            drawRun(graphics, font, shimmered, baseX, y, shadow);
        }
    }

    private static @NotNull ShimmerState shimmerState() {
        SHIMMER_CLOCK.update();
        float time = (float) SHIMMER_CLOCK.time();
        float scaleA = (float) SHIMMER_CLOCK.scaleFrom();
        float scaleB = (float) SHIMMER_CLOCK.scaleTo();
        float scaleMix = (float) SHIMMER_CLOCK.scaleMix();
        float strength = (float) SHIMMER_CLOCK.strength();
        int accent = RenderArgs.palette().accent() | 0xFF000000;
        int highlight = lerp(accent, 0xFFFFFF, 0.25F);
        return new ShimmerState(time, scaleA, scaleB, scaleMix, strength, highlight);
    }

    private static int shimmerColor(@NotNull Style style, float baseX, int index, @NotNull ShimmerState state) {
        int base = style.getColor() == null ? 0xFFFFFF : style.getColor().getValue();
        float phaseA = (baseX + index * 5.6F * state.scaleA()) * 0.075F - state.time() * 1.9F;
        float phaseB = (baseX + index * 5.6F * state.scaleB()) * 0.075F - state.time() * 1.9F;
        float waveA = 0.5F + 0.5F * (float) Math.sin(phaseA);
        float waveB = 0.5F + 0.5F * (float) Math.sin(phaseB);
        float wave = waveA + (waveB - waveA) * state.scaleMix();
        return lerp(base, state.highlight(), wave * wave * state.strength());
    }

    private static void drawRun(@NotNull GuiGraphics graphics, @NotNull Font font, @NotNull FormattedCharSequence sequence, float x, float y, boolean shadow) {
        font.drawInBatch(sequence, x, y, 0xFFFFFF, shadow, graphics.pose().last().pose(), graphics.bufferSource(), Font.DisplayMode.NORMAL, 0, 15728880);
    }

    private record ShimmerState(float time, float scaleA, float scaleB, float scaleMix, float strength, int highlight) {}

    private record CachedSequence(int[] indices, int[] codepoints, Style[] styles) implements FormattedCharSequence {
        @Override
            public boolean accept(@NotNull FormattedCharSink sink) {
                for (int i = 0; i < this.codepoints.length; i++) {
                    if (!sink.accept(this.indices[i], this.styles[i], this.codepoints[i])) {
                        return false;
                    }
                }
                return true;
            }
        }

    private static float textOffset() {
        return (float) (double) RenderArgs.TEXT_OFFSET_Y.get();
    }

    private static int lerp(int from, int to, float fraction) {
        float f = Math.max(0.0F, Math.min(1.0F, fraction));
        int r = (int) ((from >> 16 & 0xFF) + ((to >> 16 & 0xFF) - (from >> 16 & 0xFF)) * f);
        int g = (int) ((from >> 8 & 0xFF) + ((to >> 8 & 0xFF) - (from >> 8 & 0xFF)) * f);
        int b = (int) ((from & 0xFF) + ((to & 0xFF) - (from & 0xFF)) * f);
        return r << 16 | g << 8 | b;
    }

    public static @NotNull Builder builder(@NotNull Font font) {
        return new Builder(font);
    }

    public static final class Builder {
        private final Font font;
        private final List<Run> runs = new ObjectArrayList<>();
        private final IntArrayList dividers = new IntArrayList();

        private @Nullable MutableComponent pending;
        private int pendingX;
        private int cursor;

        private Builder(Font font) {
            this.font = font;
        }

        public @NotNull Builder text(@NotNull Component text) {
            if (this.pending == null) {
                this.pending = Component.empty();
                this.pendingX = this.cursor;
            }

            this.pending.append(text);
            this.cursor += this.font.width(text);
            return this;
        }

        public @NotNull Builder slot(@NotNull Component text, @NotNull String sample, @NotNull Align align) {
            this.flush();

            int textWidth = this.font.width(text);
            int slotWidth = Math.max(textWidth, this.font.width(sample));
            int x = align == Align.RIGHT ? this.cursor + slotWidth - textWidth : this.cursor;

            this.runs.add(new Run(text, text.getVisualOrderText(), x));
            this.cursor += slotWidth;
            return this;
        }

        public void divider() {
            this.flush();

            this.dividers.add(this.cursor + DIVIDER_GAP / 2);
            this.cursor += DIVIDER_GAP;
        }

        public @NotNull TextLine build() {
            this.flush();
            return new TextLine(this.runs.toArray(new Run[0]), this.dividers.toIntArray(), this.cursor);
        }

        private void flush() {
            if (this.pending == null) return;

            this.runs.add(new Run(this.pending, this.pending.getVisualOrderText(), this.pendingX));
            this.pending = null;
        }
    }
}

    public static @NotNull String memory(long megabytes) {

        return new Quantity(String.format(Locale.ROOT, "%d:%02d", whole / 3600L, whole % 3600L / 60L), " h");
    }
